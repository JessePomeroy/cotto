cmake_minimum_required(VERSION 3.20)
project(MurmurTextEngine LANGUAGES C CXX)

if(NOT APPLE)
    message(FATAL_ERROR "Murmur's local text engine requires macOS.")
endif()

# Build separately from Whisper: its ggml version and targets are independent.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_ACCELERATE ON CACHE BOOL "" FORCE)
set(GGML_BLAS ON CACHE BOOL "" FORCE)
set(GGML_BLAS_VENDOR Apple CACHE STRING "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)

set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/llama.cpp")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Run git submodule update --init --recursive first.")
endif()
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama")

add_executable(murmur-text-engine worker.cpp)
target_compile_features(murmur-text-engine PRIVATE cxx_std_17)
target_compile_options(murmur-text-engine PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(murmur-text-engine SYSTEM PRIVATE "${LLAMA_SOURCE}/vendor/nlohmann")
target_link_libraries(murmur-text-engine PRIVATE llama)
set_target_properties(murmur-text-engine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
